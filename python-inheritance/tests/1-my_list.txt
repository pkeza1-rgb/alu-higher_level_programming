Test creating an empty MyList:

>>> MyList = __import__('1-my_list').MyList

=========================
Append and print
=========================

>>> my_list.append(1)
>>> my_list.append(4)
>>> my_list.append(2)
>>> my_list.append(3)
>>> my_list.append(5)
>>> print(my_list)
[1, 4, 2, 3, 5]

=========================
print_sorted method
=========================

>>> my_list.print_sorted()
[1, 2, 3, 4, 5]

=========================
Original list unchanged
=========================

>>> print(my_list)
[1, 4, 2, 3, 5]

=========================
Different order
=========================

>>> another_list = MyList()
>>> another_list.append(10)
>>> another_list.append(3)
>>> another_list.append(7)
>>> another_list.print_sorted()
[3, 7, 10]
>>> print(another_list)
[10, 3, 7]

=========================
Negative numbers
=========================

>>> numbers = MyList()
>>> numbers.append(-5)
>>> numbers.append(2)
>>> numbers.append(-1)
>>> numbers.print_sorted()
[-5, -1, 2]
